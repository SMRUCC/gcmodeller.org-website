File

Description

NCBI GenBank database file.

The National Center for Biotechnology Information (NCBI) GenBank® database is a comprehensive, annotated collection of all publicly available nucleotide sequences and their protein translations.

文件结构说明:

  • LOCUS: 记录的摘要信息(编号/长度/分子类型/拓扑结构/分类代码/更新日期)
  • DEFINITION: 记录的描述信息
  • ACCESSION/VERSION: 记录的编号以及版本号
  • DBLINK: 关联的外部数据库编号
  • KEYWORDS: 关键词
  • SOURCE/ORGANISM: 物种信息以及分类学谱系
  • REFERENCE: 引用文献
  • COMMENT: 附加的注释说明
  • FEATURES: 位点注释数据表
  • CONTIG: CON类型的记录之中用于拼接序列的片段列表
  • ORIGIN: 序列数据

# namespace SMRUCC.genomics.Assembly.NCBI.GenBank.GBFF
export class File {
   # LocusID, GI or AccessionID
   Accession: ACCESSION;
   Comment: COMMENT;
   # CONTIG段: 对于CON类型的记录(例如人基因组的染色体)，这个字段描述了 序列是由哪些片段拼接而成的
   Contig: CONTIG;
   DbLinks: DBLINK;
   # The definition value for this organism's GenBank data.
   Definition: DEFINITION;
   Features: FEATURES;
   # 这个Genbank对象是否具有序列数据
   HasSequenceData: boolean;
   # 这个记录是否为CON(constructed)类型
   IsCON: boolean;
   # Is plasmid source?
   isPlasmid: boolean;
   # This GenBank data is the WGS(Whole genome shotgun) type data.
   IsWGS: boolean;
   Keywords: KEYWORDS;
   # The brief entry information of this genbank data.
   Locus: LOCUS;
   # This GenBank keyword section stores the sequence data for this database.
   Origin: ORIGIN;
   # PRIMARY段
   Primary: PRIMARY;
   # PROJECT段的值
   Project: string;
   # 未被显式解析的其它字段的原始文本行: 保留这些数据用于保证解析之后再写出的往返保真性
   RawSections: list;
   Reference: REFERENCE;
   # SEGMENT段的值(形如1 of 10)
   Segment: string;
   # 物种信息
   Source: SOURCE;
   # Gets the original source brief entry information of this genome.(获取这个基因组的摘要信息)
   SourceFeature: Feature;
   # the organism species name and strain information of this genbank data, this property is a combination of the source feature's species name and the source feature's strain qualifier value.
   Species: string;
   # get ncbi taxonomy id
   Taxon: string;
   # 分子的拓扑结构
   Topology: string;
   Version: VERSION;
   # WGS/WGS_SCAFLD段的值
   Wgs: string;
}
